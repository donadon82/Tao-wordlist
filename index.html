let globalBlobUrl = null; // Tự động khôi phục tiến trình làm việc cũ khi F5 document.addEventListener("DOMContentLoaded", () => { if(localStorage.getItem("intl_chu")) document.getElementById("input-chu").value = localStorage.getItem("intl_chu"); if(localStorage.getItem("intl_so")) document.getElementById("input-so").value = localStorage.getItem("intl_so"); if(localStorage.getItem("intl_kytu")) document.getElementById("input-kytu").value = localStorage.getItem("intl_kytu"); }); function toSentenceCase(str) { if (!str) return ""; const clean = str.replace(/\s+/g, ""); return clean.charAt(0).toUpperCase() + clean.slice(1).toLowerCase(); } function processInputSpace(elementId) { let val = document.getElementById(elementId).value; localStorage.setItem("intl_" + elementId.replace("input-", ""), val); val = val.replace(/,/g, " "); return val.split(/\s+/).map(item => item.trim()).filter(item => item.length > 0); } function downloadFileImmediately() { if (globalBlobUrl) { const tempLink = document.createElement("a"); tempLink.href = globalBlobUrl; tempLink.download = "generated_wordlist.txt"; document.body.appendChild(tempLink); tempLink.click(); document.body.removeChild(tempLink); } } function startGeneration() { const statusBox = document.getElementById("status-box"); const downloadLink = document.getElementById("monetized-download-link"); statusBox.style.display = "block"; downloadLink.style.display = "none"; statusBox.innerHTML = "⏳ Computing structure combinations... Please wait..."; statusBox.style.borderLeftColor = "#38bdf8"; const chuGoc = processInputSpace("input-chu"); const so = processInputSpace("input-so"); const kytu = processInputSpace("input-kytu"); if (chuGoc.length === 0 && so.length === 0 && kytu.length === 0) { statusBox.innerHTML = "⚠️ Error: Please fill in at least one input field!"; statusBox.style.borderLeftColor = "#f87171"; return; } let chuMoRong = []; chuGoc.forEach(item => { chuMoRong.push(item.toLowerCase()); chuMoRong.push(item.toUpperCase()); chuMoRong.push(toSentenceCase(item)); }); chuMoRong = [...new Set(chuMoRong)]; let wordlistSet = new Set(); // 1. Chỉ có biến thể chữ chuMoRong.forEach(c => wordlistSet.add(c)); // 2. Tổ hợp Chữ + Số chuMoRong.forEach(c => { so.forEach(s => { wordlistSet.add(c + s); wordlistSet.add(s + c); }); }); // 3. Tổ hợp Chữ + Ký tự chuMoRong.forEach(c => { kytu.forEach(k => { wordlistSet.add(c + k); wordlistSet.add(k + c); }); }); // 4. Tổ hợp đầy đủ Chữ + Số + Ký tự chuMoRong.forEach(c => { so.forEach(s => { kytu.forEach(k => { wordlistSet.add(c + s + k); wordlistSet.add(c + k + s); wordlistSet.add(s + c + k); wordlistSet.add(s + k + c); wordlistSet.add(k + c + s); wordlistSet.add(k + s + c); }); }); }); // Nếu không nhập chữ, sinh cấu trúc Số + Ký tự đặc biệt if(chuMoRong.length === 0) { so.forEach(s => { wordlistSet.add(s); kytu.forEach(k => { wordlistSet.add(s + k); wordlistSet.add(k + s); }); }); } const finalResult = Array.from(wordlistSet).join("\n"); const totalLines = wordlistSet.size; if (globalBlobUrl) { URL.revokeObjectURL(globalBlobUrl); } const blob = new Blob([finalResult], { type: "text/plain;charset=utf-8" }); globalBlobUrl = URL.createObjectURL(blob); downloadLink.style.display = "flex"; statusBox.innerHTML = `✅ Generation Complete!
Total distinct password variations generated: ${totalLines.toLocaleString()} lines.
Click the yellow button to unlock and save your file.`; statusBox.style.borderLeftColor = "var(--success-color)"; }
